import os
from flask import Flask, render_template, request, jsonify
import google.generativeai as genai
from PIL import Image
import io, base64

app = Flask(__name__)

GEMINI_API_KEY = os.environ.get("GEMINI_API_KEY", "YOUR_AQ_KEY_HERE")
genai.configure(api_key=GEMINI_API_KEY)
model = genai.GenerativeModel("gemini-3-flash-preview") # STABLE - use this, not 1.5 or 2.0

DISCLAIMERS = {
 "en": "⚠️ Disclaimer: This is not a medical diagnosis. Please consult a qualified doctor.",
 "ha": "⚠️ Gargadi: Wannan ba ganewar asibiti ba ne. Tuntuɓi likita mai lasisi.",
 "fr": "⚠️ Avertissement: Ceci n'est pas un diagnostic médical. Consultez un médecin.",
 "ar": "⚠️ إخلاء مسؤولية: هذا ليس تشخيصًا طبيًا. يرجى استشارة طبيب مؤهل."
}

SYSTEM_PROMPT = """
You are HealthCare Assistant.
Rules:
1. Reply in SAME language as user (English, Hausa, French, Arabic).
2. Provide general info, self-care, when to see doctor. NEVER diagnose.
3. If SEVERE (chest pain, can't breathe, heavy bleeding, stroke) start with 🚨 and tell to seek IMMEDIATE emergency care in their language.
4. If image provided, analyze generally but say can't definitively diagnose from image.
5. End every reply with disclaimer: {disclaimer}
Language hint: {lang}
User: {message}
"""

@app.route('/')
def index():
    return render_template('index.html')

@app.route('/api/chat', methods=['POST'])
def chat():
    try:
        data = request.json
        msg = data.get('message','')
        lang = data.get('lang','en')
        img_data = data.get('image')

        disclaimer = DISCLAIMERS.get(lang, DISCLAIMERS['en'])
        prompt = SYSTEM_PROMPT.format(disclaimer=disclaimer, lang=lang, message=msg)

        if img_data:
            if ',' in img_data: img_data = img_data.split(',')[1]
            img = Image.open(io.BytesIO(base64.b64decode(img_data)))
            response = model.generate_content([prompt, img])
        else:
            response = model.generate_content(prompt)

        return jsonify({"reply": response.text})
    except Exception as e:
        return jsonify({"reply": f"Error: {e}"}), 500

if __name__ == '__main__':
    app.run(host='0.0.0.0', port=5000, debug=True)
